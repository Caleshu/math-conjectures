```cpp
#include <boost/multiprecision/cpp_int.hpp>
#include <cstdint>
#include <iostream>

using boost::multiprecision::cpp_int;
using u64 = std::uint64_t;
using u128 = unsigned __int128;

template<class Product>
bool divides_mersenne(u64 exponent, u64 modulus) {
    u64 residue = 1;
    u64 base = 2;
    while (exponent != 0) {
        if (exponent & 1) {
            residue = static_cast<u64>(Product(residue) * base % modulus);
        }
        exponent >>= 1;
        if (exponent != 0) {
            base = static_cast<u64>(Product(base) * base % modulus);
        }
    }
    return residue == 1;
}

bool passes(u64 m) {
    if (m == 0) return true;
    if (m % 4 != 0) return false;
    return divides_mersenne<u64>(m, m - 1)
        && divides_mersenne<u64>(m, m + 1)
        && divides_mersenne<u128>(m, m * m + 1);
}

bool literal(u64 m) {
    cpp_int divisor = cpp_int(m) * m * m * m;
    if (divisor != 0) --divisor;
    cpp_int dividend = (cpp_int(1) << m) - 1;
    if (divisor == 0) return dividend == 0;
    return dividend % divisor == 0;
}

int main() {
    for (u64 m = 0; m <= 10000; ++m) {
        if (passes(m) != literal(m)) {
            std::cerr << "cross-check mismatch at m=" << m << '\n';
            return 1;
```

```cpp
#include <boost/multiprecision/cpp_int.hpp>
#include <cstdint>
#include <iostream>

using u64 = std::uint64_t;
using u128 = unsigned __int128;
using boost::multiprecision::cpp_int;

template<class Word>
bool two_power_is_one(u64 exponent, Word modulus) {
    Word base = Word(2) % modulus;
    Word value = Word(1) % modulus;
    while (exponent != 0) {
        if (exponent & 1) value = (value * base) % modulus;
        exponent >>= 1;
        if (exponent != 0) base = (base * base) % modulus;
    }
    return value == 1;
}

// For positive m. The modular tests are reached only for m >= 4.
unsigned stage(u64 m) {
    if (m % 4 != 0) return 0;
    if (!two_power_is_one<u64>(m, m - 1)) return 0;
    if (!two_power_is_one<u64>(m, m + 1)) return 1;
    if (!two_power_is_one<u128>(m, u128(m) * m + 1)) return 2;
    return 3;
}

int main() {
    for (u64 m = 0; m <= 10000; ++m) {
        const cpp_int x = m;
        cpp_int divisor = x * x * x * x;
        if (divisor != 0) --divisor;
        const cpp_int rhs = (cpp_int(1) << m) - 1;
        const bool literal = divisor == 0 ? rhs == 0 : rhs % divisor == 0;
        const bool filtered = m == 0 || stage(m) == 3;
        if (literal != filtered) {
            std::cerr << "cross-check mismatch at m=" << m << '\n';
            return 1;
        }
    }
    std::cout << "literal cross-check passed for 0..10000\n";

    u64 visited = 0, pass_minus = 0, pass_both = 0, hits = 0;
    for (u64 m = 10000004; m <= 1000000000; m += 4) {
        ++visited;
        const unsigned s = stage(m);
        pass_minus += (s >= 1);
        pass_both += (s >= 2);
        if (s == 3) {
            ++hits;
            std::cout << "solution m=" << m << '\n';
        }
    }
    if (visited != 247500000) return 2;
    std::cout << "visited=" << visited << " pass_minus=" << pass_minus
              << " pass_both=" << pass_both << " hits=" << hits << '\n';
    return 0;
}
```
