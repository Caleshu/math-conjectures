from functools import cache
from math import isqrt
from time import monotonic

params = json.load(open(os.environ['MRL_PARAMS']))


def is_prime(n):
    return n >= 2 and all(n % d for d in range(2, isqrt(n) + 1))


@cache
def residue_filter(q):
    # A modulus of the form a*q+1 often removes most candidates.
    for a in range(1, 65):
        modulus = a * q + 1
        if is_prime(modulus):
            residues = {pow(b, q, modulus) for b in range(modulus)}
            return modulus, residues
    return 1, {0}  # Pass all integers to the exact root test.


@cache
def tests_for_bits(bits):
    return [(q, *residue_filter(q)) for q in range(2, bits) if is_prime(q)]


def power_witness(v):
    if v < 4:
        return None
    bits = v.bit_length()
    for q, modulus, residues in tests_for_bits(bits):
        if v % modulus not in residues:
            continue
        if q == 2:
            base = isqrt(v)
        else:
            low = 1
            high = 1 << ((bits + q - 1) // q)
            while high - low > 1:
                mid = (low + high) // 2
                if mid ** q <= v:
                    low = mid
                else:
                    high = mid
            base = low
        if base > 1 and base ** q == v:
            return base, q
    return None


def main():
    deadline = monotonic() + 105
    n_max = params['n_max']
    if n_max < 0:
        raise ValueError('n_max must be nonnegative')

    small_limit = 512
    direct_powers = set()
    for base in range(2, isqrt(small_limit) + 1):
        value = base * base
        while value <= small_limit:
            direct_powers.add(value)
            value *= base
    power_errors = []
    for value in range(1, small_limit + 1):
        witness = power_witness(value)
        if (witness is not None) != (value in direct_powers):

import json, os
from functools import lru_cache
from itertools import combinations
from time import monotonic

params = json.load(open(os.environ['MRL_PARAMS']))


def shifted_partitions(bound):
    last = 16
    while True:
        counts = [1] + [0] * last
        for part in range(1, last + 1):
            for total in range(part, last + 1):
                counts[total] += counts[total - part]
        if counts[last] + 2 > bound:
            return sorted({p + 2 for p in counts[1:] if p + 2 <= bound})
        last *= 2


def attainable(values, bound, deadline):
    size = len(values)
    edges = [0] * size
    for i in range(size):
        for j in range(i + 1, size):
            if values[j] % values[i] == 0:
                edges[i] |= 1 << j
                edges[j] |= 1 << i
    limit = (1 << (bound + 1)) - 1

    @lru_cache(maxsize=256)
    def sums(mask):
        if monotonic() >= deadline:
            raise TimeoutError
        if not mask:
            return 1
        isolated = 0
        highest_degree = 0
        pivot = -1
        remaining = mask
        while remaining:
            bit = remaining & -remaining
            i = bit.bit_length() - 1
            degree = (edges[i] & mask).bit_count()
            if degree == 0:
                isolated |= bit
            elif degree > highest_degree:
                highest_degree = degree
                pivot = i
            remaining ^= bit
        if isolated:
            reachable = sums(mask ^ isolated)
            while isolated:
                bit = isolated & -isolated
                weight = values[bit.bit_length() - 1]
                reachable |= (reachable << weight) & limit
                isolated ^= bit
            return reachable
        rest = mask ^ (1 << pivot)
        without_pivot = sums(rest)
        with_pivot = sums(rest & ~edges[pivot]) << values[pivot]
        return without_pivot | (with_pivot & limit)

    try:
        return sums((1 << size) - 1)
    finally:
        sums.cache_clear()


def direct_small_sums():
    # p(1)+2 through p(9)+2. The next value is p(10)+2 = 44.
    small = [3, 4, 5, 7, 9, 13, 17, 24, 32]
    totals = set()
    for length in range(len(small) + 1):
        for selected in combinations(small, length):
            total = sum(selected)
            if total <= 40 and all(b % a != 0 for a, b in combinations(selected, 2)):
                totals.add(total)
    return totals


def main():
    deadline = monotonic() + 105
    bounds = sorted(set(params['bounds']))
    all_values = shifted_partitions(max([7141, *bounds]))
    small_values = [v for v in all_values if v <= 40]
    small_bits = attainable(small_values, 40, deadline)
    direct = direct_small_sums()
    mismatches = [t for t in range(41) if bool(small_bits & (1 << t)) != (t in direct)]

    witness = [137, 178, 299, 1960, 4567]
    valid = (sum(witness) == 7141 and all(v in all_values for v in witness)

        missing = [t for t in range(3, bound + 1) if not (packed[t // 8] & (1 << (t % 8)))]
        checks.append({'bound': bound, 'missing': missing})
        eligible = len(values)

    if mismatches or not valid:

        summary = f'Partial run completed {len(checks)} of {len(bounds)} exact coverage checks.'
    else:
        summary = f'Completed {len(checks)} of {len(bounds)} exact coverage checks.'
    return {'summary': summary, 'checks': checks,
            'eligible_at_largest_completed_bound': eligible,
            'direct_subset_check': {'range': [0, 40], 'mismatches': mismatches},
            'witness_7141': {'summands': witness, 'valid': valid}}


with open('result.json', 'w') as output:
    json.dump(main(), output)

def is_prime(n):
    return n >= 2 and all(n % d for d in range(2, isqrt(n) + 1))


@cache
def residue_filter(q):
    # A modulus of the form a*q+1 often removes most candidates.
    for a in range(1, 65):
        modulus = a * q + 1
        if is_prime(modulus):
            residues = {pow(b, q, modulus) for b in range(modulus)}
            return modulus, residues
    return 1, {0}  # Pass all integers to the exact root test.


@cache
def tests_for_bits(bits):
    return [(q, *residue_filter(q)) for q in range(2, bits) if is_prime(q)]


def power_witness(v):
    if v < 4:
        return None
    bits = v.bit_length()
    for q, modulus, residues in tests_for_bits(bits):
        if v % modulus not in residues:
            continue
        if q == 2:
            base = isqrt(v)
        else:
            low = 1
            high = 1 << ((bits + q - 1) // q)
            while high - low > 1:
                mid = (low + high) // 2
                if mid ** q <= v:
                    low = mid
                else:
                    high = mid
            base = low
        if base > 1 and base ** q == v:
            return base, q
    return None

def attainable(values, bound, deadline):
    size = len(values)
    edges = [0] * size
    for i in range(size):
        for j in range(i + 1, size):
            if values[j] % values[i] == 0:
                edges[i] |= 1 << j
                edges[j] |= 1 << i
    limit = (1 << (bound + 1)) - 1

    @lru_cache(maxsize=256)
    def sums(mask):
        if monotonic() >= deadline:
            raise TimeoutError
        if not mask:
            return 1
        isolated = 0
        highest_degree = 0
        pivot = -1
        remaining = mask
        while remaining:
            bit = remaining & -remaining
            i = bit.bit_length() - 1
            degree = (edges[i] & mask).bit_count()
            if degree == 0:
                isolated |= bit
            elif degree > highest_degree:
                highest_degree = degree
                pivot = i
            remaining ^= bit
        if isolated:
            reachable = sums(mask ^ isolated)
            while isolated:
                bit = isolated & -isolated
                weight = values[bit.bit_length() - 1]
                reachable |= (reachable << weight) & limit
                isolated ^= bit
            return reachable
        rest = mask ^ (1 << pivot)
        without_pivot = sums(rest)
        with_pivot = sums(rest & ~edges[pivot]) << values[pivot]
        return without_pivot | (with_pivot & limit)

    try:
        return sums((1 << size) - 1)
    finally:
        sums.cache_clear()
